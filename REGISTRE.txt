package simulateurmotorola6809;

public class Registres {
private int PC;
private int A;
private int B;
private int X;
private int Y;
private int S;
private int U;
private int DP;
private int CCR;


public Registres(int PC, int A, int B, int X, int Y, int S, int U, int DP, int CCR) {
	this.PC = PC;
	this.A = A;
	this.B = B;
	this.X = X;
	this.Y = Y;
	this.S = S;
	this.U = U;
	this.DP = DP;
	this.CCR = CCR;
}

public int getPC() {
	return PC;
}

public int getA() {
	return A;
}
public void setA(int a) {
	A = a;
}
public int getB() {
	return B;
}
public void setB(int b) {
	B = b;
}
public int getX() {
	return X;
}
public void setX(int x) {
	X = x;
}
public int getY() {
	return Y;
}
public void setY(int y) {
	Y = y;
}
public int getS() {
	return S;
}
public void setS(int s) {
	S = s;
}
public int getU() {
	return U;
}
public void setU(int u) {
	U = u;
}
public int getDP() {
	return DP;
}
public void setDP(int dP) {
	DP = dP;
}
public int getCCR() {
	return CCR;
}
public void setCCR(int cCR) {
	CCR = cCR;
}

public void setPC(int pC) {
	PC = pC;
}





}
