	private int tailleInstructionIndexee(byte postbyte) {
		
	    int post = postbyte & 0xFF;
	    
	    if (post == 0x00 ||post == 0x80 ||post == 0x82) { 
	        return 2;
	    }
	    else if (post >= 0x88 && post <= 0x8F) {
	        return 3;
	    }
	    else if (post == 0x89) {
	        return 4;
	    }
	    else if (post == 0x8C) {
	        return 3;
	    }
		return post;
	    }
	private int calculerAdresseIndexee(byte postbyte, int pc) {
        int post = postbyte & 0xFF;
        if (post == 0x00) {
            return registres.getX();
        }
        else if (post == 0x80) {
            int adresse = registres.getX();
            registres.setX(adresse + 1);
            return adresse;
        }
        else if (post == 0x82) {
            registres.setX(registres.getX() - 1);
            return registres.getX();
        }
        else if ((post & 0x9F) == 0x88) { 
            byte offset = memoire[pc + 2];
            return registres.getX() + (offset & 0xFF);
        }
        else if (post == 0x89) {
            byte offsetMSB = memoire[pc + 2];
            byte offsetLSB = memoire[pc + 3];
            int offset = ((offsetMSB & 0xFF) << 8) | (offsetLSB & 0xFF);
            return registres.getX() + offset;
        }
		return post;
	}

	        fermerButton.addActionListener(new ActionListener() {

	    private void reinitialiserEtat() {
	        // Réinitialiser les registres
	        registres = new Registres(0x0000, 0x00, 0x00, 0x0000, 0x0000, 0x0000, 0x0000, 0x00, 0x00);
	        
	        // Réinitialiser le simulateur
	        motorola = new Motorola();
	        motorola.registres = registres;
	        
	        // Réinitialiser l'index d'instruction
	        indexInstruction = 0;
	        instructions = null;
	        
	        // Mettre à jour l'affichage
	        mettreAJourAffichage();
	        
	        JOptionPane.showMessageDialog(this, 
	            "Le simulateur a été réinitialisé à l'état initial.", 
	            "Réinitialisation", 
	            JOptionPane.INFORMATION_MESSAGE);
	    }
	    
	    private void executerInstructionUnique() {
	        String texte = editeurArea.getText().trim();
	        if (texte.isEmpty()) {
	            JOptionPane.showMessageDialog(this, "Aucune instruction à exécuter", "Erreur", JOptionPane.ERROR_MESSAGE);
	            return;
	        }
	        
	        String[] lignes = texte.split("\n");
	        String instruction = lignes[0].trim();
	        
	        if (executerUneInstruction(instruction)) {
	            JOptionPane.showMessageDialog(this, 
	                "Instruction exécutée : " + instruction + 
	                "\n\nVérifiez les registres!", 
	                "Exécution réussie", 
	                JOptionPane.INFORMATION_MESSAGE);
	        }
	    }
	    
	    private boolean executerUneInstruction(String instruction) {
	        String[] parties = instruction.split("\\s+", 2);
	        
	        if (parties.length < 2) {
	            JOptionPane.showMessageDialog(this, 
	                "Format incorrect : " + instruction + 
	                "\nExemple : LDA #$15", 
	                "Erreur", 
	                JOptionPane.ERROR_MESSAGE);
	            return false;
	        }
	        
	        String mnemonic = parties[0].trim().toUpperCase();
	        String operande = parties[1].trim();
	        
	        try {
	            byte[] codeMachine = convertisseur.convertir(mnemonic, operande);
	            
	            if (codeMachine.length == 0) {
	                JOptionPane.showMessageDialog(this, 
	                    "Instruction non supportée : " + mnemonic, 
	                    "Erreur", 
	                    JOptionPane.ERROR_MESSAGE);
	                return false;
	            }
	            
	            motorola.chargerEnMemoire(codeMachine);
	            motorola.executerInstruction();
	            
	            mettreAJourAffichage();
	            return true;
	            
	        } catch (Exception e) {
	            JOptionPane.showMessageDialog(this, 
	                "Erreur : " + e.getMessage(), 
	                "Erreur d'exécution", 
	                JOptionPane.ERROR_MESSAGE);
	            return false;
	        }
	    }
	    
	    private void preparerPasAPas() {
	        String texte = editeurArea.getText().trim();
	        if (texte.isEmpty()) {
	            JOptionPane.showMessageDialog(this, "Aucune instruction à exécuter", "Erreur", JOptionPane.ERROR_MESSAGE);
	            return;
	        }
	        
	        instructions = texte.split("\n");
	        indexInstruction = 0;
	        
	        JOptionPane.showMessageDialog(this, 
	            instructions.length + " instructions chargées.\n" +
	            "Cliquez sur 'Suivant' pour exécuter chaque instruction.", 
	            "Prêt pour pas à pas", 
	            JOptionPane.INFORMATION_MESSAGE);
	        
	        executerPasAPasAvecDialogue();
	    }
	    
	    private void executerPasAPasAvecDialogue() {
	        if (indexInstruction >= instructions.length) {
	            JOptionPane.showMessageDialog(this, 
	                "Toutes les instructions ont été exécutées!", 
	                "Terminé", 
	                JOptionPane.INFORMATION_MESSAGE);
	            return;
	        }
	        
	        String instruction = instructions[indexInstruction].trim();
	        if (instruction.isEmpty()) {
	            indexInstruction++;
	            executerPasAPasAvecDialogue();
	            return;
	        }
	        
	        if (executerUneInstruction(instruction)) {
	            int choix = JOptionPane.showConfirmDialog(this, 
	                "Instruction " + (indexInstruction + 1) + "/" + instructions.length + " exécutée : " + instruction + 
	                "\n\nContinuer ?", 
	                "Pas à pas", 
	                JOptionPane.YES_NO_OPTION);
	            
	            indexInstruction++;
	            
	            if (choix == JOptionPane.YES_OPTION && indexInstruction < instructions.length) {
	                executerPasAPasAvecDialogue();
	            }
	        }
	    }

	    private void afficherZoneMemoire(String titre, int adresseDebut, int adresseFin) {
	        JDialog dialog = new JDialog(this, titre, true);
	        dialog.setSize(700, 500);
	        dialog.setLayout(new BorderLayout());
	        
	        JTextArea memoireArea = new JTextArea(25, 60);
	        memoireArea.setFont(new Font("Monospaced", Font.PLAIN, 11));
	        memoireArea.setEditable(false);
	        
	        StringBuilder sb = new StringBuilder();
	        sb.append("Adresse  Valeur\n");
	        sb.append("════════════════\n");
	        
	        int limite = Math.min(adresseFin, adresseDebut + 0x200);
	        
	        for (int adresse = adresseDebut; adresse < limite; adresse++) {
	            byte valeur = motorola.memoire[adresse];
	            sb.append(String.format("%04X     %02X\n", adresse, valeur & 0xFF));
	        }
	        
	        if (adresseFin > limite) {
	            sb.append("\n... (affichage limité) ...\n");
	        }
	        
	        memoireArea.setText(sb.toString());
	        
	        JScrollPane scrollPane = new JScrollPane(memoireArea);
	        dialog.add(scrollPane, BorderLayout.CENTER);
	        
	        JPanel panelBoutons = new JPanel();
	        JButton fermerButton = new JButton("Fermer");
	        fermerButton.addActionListener(new ActionListener() {
	            @Override
	            public void actionPerformed(ActionEvent e) {
	                dialog.dispose();
	            }
	        });
	        
	        panelBoutons.add(fermerButton);
	        dialog.add(panelBoutons, BorderLayout.SOUTH);
	        
	        dialog.setLocationRelativeTo(this);
	        dialog.setVisible(true);
	    }
	    
	    private void mettreAJourAffichage() {
	        // Mise à jour des registres
	        pcField.setText(String.format("%04X", registres.getPC()));
	        aField.setText(String.format("%02X", registres.getA()));
	        bField.setText(String.format("%02X", registres.getB()));
	        xField.setText(String.format("%04X", registres.getX()));
	        yField.setText(String.format("%04X", registres.getY()));
	        sField.setText(String.format("%04X", registres.getS()));
	        uField.setText(String.format("%04X", registres.getU()));
	        dpField.setText(String.format("%02X", registres.getDP()));
	        
	        // Mise à jour du CCR
	        int ccr = registres.getCCR();
	        eCheckBox.setSelected((ccr & 0x80) != 0);
	        fCheckBox.setSelected((ccr & 0x40) != 0);
	        hCheckBox.setSelected((ccr & 0x20) != 0);
	        iCheckBox.setSelected((ccr & 0x10) != 0);
	        nCheckBox.setSelected((ccr & 0x08) != 0);
	        zCheckBox.setSelected((ccr & 0x04) != 0);
	        vCheckBox.setSelected((ccr & 0x02) != 0);
	        cCheckBox.setSelected((ccr & 0x01) != 0);
	        
	        // Mise à jour de la RAM
	        StringBuilder ramText = new StringBuilder();
	        ramText.append("Adresse  Valeur\n");
	        for (int i = 0x0000; i < 0x000C; i++) {
	            ramText.append(String.format("%04X     %02X\n", i, motorola.memoire[i] & 0xFF));
	        }
	        ramArea.setText(ramText.toString());
	        
	        // Mise à jour de la ROM
	        StringBuilder romText = new StringBuilder();
	        romText.append("Adresse  Valeur\n");
	        for (int i = 0xC000; i < 0xC00C; i++) {
	            romText.append(String.format("%04X     %02X\n", i, motorola.memoire[i] & 0xFF));
	        }
	        romArea.setText(romText.toString());
	    }
